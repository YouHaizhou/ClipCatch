## 问题 3A：YouTube 搜索报 "All connections attempt failed"

**文件**：`src/services/search_service.py`，函数 `_search_youtube`

**根因**：直接请求 `youtube.com` 在国内网络环境下会被拦截/超时，需要改为通过 Serper API 搜索 YouTube。

**修复方案**：优先用 Serper API 搜索 YouTube，无 Serper Key 时降级为直连（并给出友好错误）：

```python
async def _search_youtube(query: str, page: int = 1) -> list:
    """YouTube 搜索：优先 Serper API，无 Key 时尝试直连"""
    # 1. 尝试 Serper API（国内可用，推荐）
    serper_results = await _search_youtube_via_serper(query, page)
    if serper_results is not None:
        return serper_results
    # 2. 降级：直连 YouTube（需要科学上网）
    return await _search_youtube_direct(query, page)


async def _search_youtube_via_serper(query: str, page: int = 1):
    """通过 Serper API 搜索 YouTube 视频，返回 None 表示未配置"""
    from database import SessionLocal
    from models import Setting
    import json as _json
    db = SessionLocal()
    try:
        row = db.query(Setting).filter(Setting.key == 'api_key_serper').first()
        serper_key = _json.loads(row.value) if row and row.value else None
    finally:
        db.close()
    if not serper_key:
        return None  # 未配置，让调用方降级
    try:
        async with httpx.AsyncClient(timeout=15) as client:
            resp = await client.post(
                'https://google.serper.dev/videos',
                headers={'X-API-KEY': serper_key, 'Content-Type': 'application/json'},
                json={'q': f'{query} site:youtube.com', 'num': 12},
            )
        if resp.status_code != 200:
            return None
        data = resp.json()
        results = []
        for item in data.get('videos', []):
            link = item.get('link', '')
            if 'youtube.com/watch' not in link and 'youtu.be' not in link:
                continue
            # 从 URL 提取 video_id
            vid_match = re.search(r'(?:v=|youtu\.be/)([\w-]{11})', link)
            video_id = vid_match.group(1) if vid_match else link
            duration_str = item.get('duration', '')
            results.append({
                'id': video_id,
                'title': item.get('title', ''),
                'url': link,
                'platform': 'youtube',
                'duration': _parse_duration(duration_str),
                'thumbnail_url': item.get('imageUrl', ''),
                'author': item.get('channel', ''),
                'published_at': item.get('date', ''),
                'view_count': 0,
            })
        return results
    except Exception:
        return None


async def _search_youtube_direct(query: str, page: int = 1) -> list:
    """直连 YouTube（需科学上网），原有实现"""
    search_url = f'https://www.youtube.com/results?search_query={query.replace(" ", "+")}'
    headers = {**_HEADERS, 'Accept-Language': 'en-US,en;q=0.9'}
    try:
        async with httpx.AsyncClient(timeout=15, follow_redirects=True) as client:
            resp = await client.get(search_url, headers=headers)
        if resp.status_code != 200:
            raise ConnectionError(f'YouTube 返回 HTTP {resp.status_code}')
        match = re.search(r'var ytInitialData = (\{.+?\});', resp.text, re.DOTALL)
        if not match:
            raise ValueError('无法解析 YouTube 搜索结果')
        data = json.loads(match.group(1))
        contents = data['contents']['twoColumnSearchResultsRenderer']['primaryContents']['sectionListRenderer']['contents']
        results = []
        for section in contents:
            for item in section.get('itemSectionRenderer', {}).get('contents', []):
                vr = item.get('videoRenderer')
                if not vr: continue
                video_id = vr.get('videoId', '')
                title = ''.join(r.get('text', '') for r in vr.get('title', {}).get('runs', []))
                duration_sec = _parse_duration(vr.get('lengthText', {}).get('simpleText', ''))
                thumbs = vr.get('thumbnail', {}).get('thumbnails', [])
                thumb = thumbs[-1]['url'] if thumbs else ''
                author = ''.join(r.get('text', '') for r in vr.get('ownerText', {}).get('runs', []))
                if video_id and title:
                    results.append({'id': video_id, 'title': title,
                        'url': f'https://www.youtube.com/watch?v={video_id}',
                        'platform': 'youtube', 'duration': duration_sec,
                        'thumbnail_url': thumb, 'author': author,
                        'published_at': '', 'view_count': 0})
                if len(results) >= 12: break
            if len(results) >= 12: break
        return results
    except Exception as e:
        raise ConnectionError(f'YouTube 直连失败（建议配置 Serper API Key）: {e}')
```

## 问题 3B：Twitter 搜索「未找到相关视频」

**根因**：Serper 的 `/search` 接口返回的是网页搜索结果，Twitter 链接过滤条件太严，且需要用 `/videos` 接口专门搜索视频内容。

**修复**：`_search_twitter` 改用 Serper `/videos` 接口，并放宽筛选条件：

```python
async def _search_twitter(query: str, page: int = 1) -> list:
    from database import SessionLocal
    from models import Setting
    import json as _json
    db = SessionLocal()
    try:
        row = db.query(Setting).filter(Setting.key == 'api_key_serper').first()
        serper_key = _json.loads(row.value) if row and row.value else None
    finally:
        db.close()
    if not serper_key:
        return []
    try:
        # 先用 /videos 接口
        async with httpx.AsyncClient(timeout=15) as client:
            resp = await client.post(
                'https://google.serper.dev/videos',
                headers={'X-API-KEY': serper_key, 'Content-Type': 'application/json'},
                json={'q': f'site:twitter.com {query}', 'num': 10},
            )
        results = []
        if resp.status_code == 200:
            for item in resp.json().get('videos', []):
                link = item.get('link', '')
                results.append({
                    'id': link.split('/')[-1] or link,
                    'title': item.get('title', ''),
                    'url': link,
                    'platform': 'twitter',
                    'duration': _parse_duration(item.get('duration', '')),
                    'thumbnail_url': item.get('imageUrl', ''),
                    'author': item.get('channel', ''),
                    'published_at': item.get('date', ''),
                    'view_count': 0,
                })
        # 若 /videos 无结果，再用 /search 补充
        if not results:
            async with httpx.AsyncClient(timeout=15) as client:
                resp2 = await client.post(
                    'https://google.serper.dev/search',
                    headers={'X-API-KEY': serper_key, 'Content-Type': 'application/json'},
                    json={'q': f'site:twitter.com OR site:x.com {query}', 'num': 10},
                )
            if resp2.status_code == 200:
                for item in resp2.json().get('organic', []):
                    link = item.get('link', '')
                    if not link: continue
                    results.append({
                        'id': link.split('/')[-1] or link,
                        'title': item.get('title', ''),
                        'url': link,
                        'platform': 'twitter',
                        'duration': 0,
                        'thumbnail_url': item.get('imageUrl', ''),
                        'author': '',
                        'published_at': item.get('date', ''),
                        'view_count': 0,
                    })
        return results[:12]
    except Exception:
        return []
```
